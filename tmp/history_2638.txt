+function analisarFolhaTemplateProjeto(nome,rows){
+  const aliases={
+    poste:['poste','numero poste','n poste','codigo poste'],
+    codigo:['codigo','cod','desenho','referencia','ref','material'],
+    nome:['nome','descricao','designacao'],
+    quantidade:['quantidade','qtd','qt','qty'],
+    observacoes:['observacoes','observacao','obs']
+  };
+  let melhor={linha:-1,mapa:{},pontos:0};
+  rows.slice(0,12).forEach((row,idx)=>{
+    const mapa={};let pontos=0;
+    (row||[]).forEach((v,col)=>{
+      const h=normalizarCabecalhoProjeto(v);
+      for(const [campo,lista] of Object.entries(aliases)){
+        if(lista.includes(h)&&mapa[campo]==null){mapa[campo]=col;pontos++;}
+      }
+    });
+    if(pontos>melhor.pontos)melhor={linha:idx,mapa,pontos};
+  });
+  const n=normalizarCabecalhoProjeto(nome);
+  const tipo=n.includes('poste')?'Postes':n.includes('desenho')?'Desenhos':n.includes('mater')?'Materiais':n.includes('template')?'Template':'Outra';
+  if(melhor.linha<0)return {nome,tipo,total:0,validas:0,erros:['Não foi possível identificar os cabeçalhos.'],colunas:[]};
+  const dados=rows.slice(melhor.linha+1).filter(r=>(r||[]).some(v=>String(v||'').trim()));
+  const erros=[];let validas=0;
+  dados.forEach((r,i)=>{
+    const chave=tipo==='Postes'?(melhor.mapa.poste!=null?r[melhor.mapa.poste]:''):(melhor.mapa.codigo!=null?r[melhor.mapa.codigo]:'');
+    const qtd=melhor.mapa.quantidade!=null?r[melhor.mapa.quantidade]:'';
+    if(!String(chave||'').trim())erros.push('Linha '+(melhor.linha+i+2)+': falta '+(tipo==='Postes'?'o número do poste':'o código/referência')+'.');
+    else if(String(qtd||'').trim()!==''&&(!Number.isFinite(Number(qtd))||Number(qtd)<0))erros.push('Linha '+(melhor.linha+i+2)+': quantidade inválida.');
+    else validas++;
+  });
+  return {nome,tipo,total:dados.length,validas,erros,colunas:Object.keys(melhor.mapa)};
